void print (const char *text)
 Accepts text in brackets("), that is later outputted as a new line in the output console, that can be viewed if you use "idf.py monitor" in ESP-IDF terminal.
void print_int (int value)
 Same as print(), but only accepts and prints integer values.
void sleep_ms (int ms)
 Accepts integer values, waits / stops the current process for the specified time, measured in milliseconds.
int ticks_ms (void)
 Returns the amount of time passed since your program started.
uint16_t bios_rgb565 (int r, int g, int b)
 Translates standard RGB values into the color palette / scheme of the screen.
void gfx_present (void)
 Your program draws an image in a buffer, this command submits it to the screen.
void gfx_size (int *width, int *height)
 Returns the display size, first the width, then the height, measured in pixels.
void gfx_clear (uint16_t rgb565)
 Fills the entire screen with a solid color.
void gfx_fill_rect (int x, int y, int w, int h, uint16_t rgb565)
 Draws a rectangle on specified X and Y coordinates (top-left origin), with a specified width (W) and height (H).
void gfx_pixel (int x, int y, uint16_t rgb565)
 Draws a singular pixel on specified X and Y coordinates.
void gfx_backlight (int level)
 Sets the screen's backlight / brightness to a specified amount.
void gfx_draw_text (int x, int y, uint16_t rgb565)
 Draws text on the screen, starting from a specified X and Y coordinate (top-left origin), with a specified color.
void gfx_draw_text_scaled (int x, int y, uint16_t rgb565, int scale)
 Same as gfx_draw_text(), but this command can scale text up by a whole (integer) amount.
int gfx_text_width (const char *text)
 Returns the text width of the specified text, measured in pixels.
int gfx_text_height (const char *text)
 Same as gfx_text_width(), but returns height of the text.
void gfx_draw_image (int x, int y, int asset_id)
 Draws an image from a specified X and Y coordinate (top-left origin), from an asset id you got when loading that image.
void select_game (const char *name)

int read_self (const char *path, char *buffer, int size)
 Reads from "sd/GAMES/<this_app_name>/...".
int read_data (const char *path, char *buffer, int size)

int list_self (const char *dir, char *buffer, int size)
 Same as read_self() / read_data(), but lists files from "sd/GAMES/<this_app_name>/...".
int list_data (const char *dir, char *buffer, int size)
 Same as list_self(), but lists files from "sd/ASSETS/<this_app_name>/...".
int asset_load_self (const char *path)
 Loads an .eimg image from "sd/GAMES/<this_app_name>/...".
int asset_load_data (const char *path)
 Same as asset_load_self(), but the starting point for that relative path changes to "sd/ASSETS/<this_app_name>/...".
void asset_uload (int asset_id)

int asset_defragment (void)
 Rebuilds currently loaded assets, repacks the asset pool while keeping the same asset IDs. Defragments the asset pool.
int button_pressed (const char *name)
 Returns true if the passed button is pressed (active).
void install_game (const char *name)

void start_installed_game (void)
